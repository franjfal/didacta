% Lección: el determinante (castellano).

\begin{frame}
\didactatitle{El determinante de una matriz cuadrada}

\begin{definition}[Orden dos]
\[
  \det \begin{pmatrix} a & b \\ c & d \end{pmatrix} = ad - bc .
\]
\end{definition}

\begin{proposition}
$A$ tiene inversa si y solo si $\det A \neq 0$.
\end{proposition}
\end{frame}

\begin{frame}
\didactatitle{El producto de matrices}

\begin{definition}[Producto]
Si $A$ es de tamaño $m \times n$ y $B$ de tamaño $n \times p$, el producto
$AB$ es la matriz $m \times p$ cuya entrada $(i, j)$ es
\[
  (AB)_{ij} = \sum_{k=1}^{n} a_{ik}\, b_{kj} .
\]
\end{definition}

\begin{example}
\[
  \begin{pmatrix} 1 & 2 \\ 0 & 1 \end{pmatrix}
  \begin{pmatrix} 3 \\ 4 \end{pmatrix}
  = \begin{pmatrix} 11 \\ 4 \end{pmatrix} .
\]
\end{example}
\end{frame}

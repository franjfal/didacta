% Problema: el límite de una raíz (castellano).

\begin{exercise}[El límite de una raíz]
Calcula $\displaystyle \lim_{x \to 4} \frac{\sqrt{x} - 2}{x - 4}$.

\dmarks{2}

\begin{answer}
$1/4$.
\end{answer}

\begin{solution}
Multiplicando y dividiendo por $\sqrt{x} + 2$, el cociente es
$1/(\sqrt{x} + 2)$, que tiende a $1/4$.
\end{solution}
\end{exercise}
